Código Python Oficial (Base IPEA)

Copie o código abaixo para rodar no Google Colab ou no seu terminal Python:

import urllib.request
import csv
import io

URL_FONTE = "https://mapaosc.ipea.gov.br/download/20260522_MOSC_baseDivulgacao.csv"
ARQUIVO_SAIDA = "planilha_osc_1000_registros.csv"
META_REGISTROS = 1000

print(f"Buscando {META_REGISTROS} registros no servidor IPEA...")
req = urllib.request.Request(URL_FONTE, headers={'User-Agent': 'Mozilla/5.0'})

try:
    with urllib.request.urlopen(req) as resp:
        txt_stream = io.TextIOWrapper(resp, encoding='utf-8', errors='ignore')
        reader = csv.reader(txt_stream, delimiter=';')
        next(reader)

        with open(ARQUIVO_SAIDA, mode='w', newline='', encoding='utf-8-sig') as f:
            writer = csv.writer(f, delimiter=';')
            writer.writerow(['CNPJ', 'RAZÃO SOCIAL', 'NATUREZA JURÍDICA', 'UF'])

            count = 0
            for row in reader:
                if len(row) > 4:
                    writer.writerow([row[0], row[1], row[2], row[4]])
                    count += 1
                    if count >= META_REGISTROS:
                        break
            print(f"Sucesso! {count} registros salvos em {ARQUIVO_SAIDA}")
except Exception as e:
    print(f"Erro ao processar: {e}")
